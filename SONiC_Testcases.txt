def clean_number(num_str):
    return int(num_str.replace(",", "").strip())


def parse_counters(output):
    """
    Parse the output of 'show interface counters' and return a dictionary with RX_OK and TX_OK.
    """
    counters = {}
    lines = output.splitlines()
    header_found = False

    for line in lines:
        if re.match(r'^\s*IFACE\s+', line):
            header_found = True
            continue
        if not header_found or not line.strip():
            continue

        #for matching the lines in TX/RX
        match = re.match(r'^\s*(Ethernet\d+)\s+\S+\s+([\d,]+)\s+.*?\s+([\d,]+)\s+', line)
        if match:
            iface = match.group(1)
            rx_ok = clean_number(match.group(2))
            tx_ok = clean_number(match.group(3))
            counters[iface] = {
                'rx_ok': rx_ok,
                'tx_ok': tx_ok
            }

    return counters

def interface_list(uut1, uut1_int1, uut1_int2, interface_list, min_delta=1000, detailed_check=False):
    logger.info("Collecting initial interface counters...")
    if detailed_check:
        before = {}
        for intf in interface_list:
            output = uut1.execute(f'show interface counters -i "{intf}"')
            before[intf] = parse_counters(output).get(intf, {})
    else:
        before = parse_counters(uut1.execute("show interface counters"))

    time.sleep(30)

    logger.info("Collecting post-wait interface counters...")
    if detailed_check:
        after = {}
        for intf in interface_list:
            output = uut1.execute(f'show interface counters -i "{intf}"')
            after[intf] = parse_counters(output).get(intf, {})
    else:
        after = parse_counters(uut1.execute("show interface counters"))

    all_good = True
    for intf in interface_list:
        rx1 = int(before.get(intf, {}).get('rx_ok', 0))
        tx1 = int(before.get(intf, {}).get('tx_ok', 0))
        rx2 = int(after.get(intf, {}).get('rx_ok', 0))
        tx2 = int(after.get(intf, {}).get('tx_ok', 0))

        delta_rx = rx2 - rx1
        delta_tx = tx2 - tx1

        logger.info(
            f"[{intf}] RX: {rx1:,} → {rx2:,} (Δ={delta_rx:,}) | "
            f"TX: {tx1:,} → {tx2:,} (Δ={delta_tx:,})"
        )

        if delta_rx < min_delta or delta_tx < min_delta:
            logger.warning(f"{intf} failed min traffic threshold. RX: {delta_rx}, TX: {delta_tx}")
            all_good = False

    return all_good


def flap_and_check(uut1, uut1_int1, uut1_int2, interface_list, iterations=100):
    logger.info(f"Flapping interfaces {iterations} times and checking traffic...")

    enter_root_shell(uut1)

    for i in range(iterations):
        logger.info(f"--- Flap Iteration {i+1} ---")
        for intf in interface_list:
            uut1.sendline(f"config interface shutdown {intf}")
            uut1.expect("#")
            time.sleep(2)

            uut1.sendline(f"config interface startup {intf}")
            uut1.expect("#")
            time.sleep(5)

    exit_root_shell(uut1)

    return check_interface_traffic(uut1, interface_list)

def check_interface_traffic(uut1, uut1_int1, uut1_int2, interface_list):

    logger.info("Traffic check passed")
    return True

# Normalize data for comparison
def normalize(eeprom_str):
    """Extract stable fields for EEPROM comparison."""
    fields_to_check = ["Serial Number", "Vendor", "Part Number"]
    normalized = []

    for line in eeprom_str.splitlines():
        for field in fields_to_check:
            if field in line:
                # Normalize whitespace and format
                normalized.append(re.sub(r'\s+', ' ', line.strip()))
    return "\n".join(sorted(normalized))

            }
        else:
            logger.info(f"Unknown direction: {direction}")
            exit(1)

        # Validate thresholds
        status = True
        for sensor in expected:
            actual = temp_table.get(sensor)
            if not actual:
                logger.info(f"{sensor} data not found in output")
                status = False
                continue

            if actual == expected[sensor]:
                logger.info(f"Temperature thresholds on {sensor} matched: "
                            f"Expected: {{High: {expected[sensor]['high_th']}, Critical High: {expected[sensor]['crit_high_th']}}}, "
                            f"Found: {{High: {actual['high_th']}, Critical High: {actual['crit_high_th']}}}")

                logger.warning(f"Temperature thresholds on {sensor} mismatched: "
                            f"Expected: {{High: {expected[sensor]['high_th']}, Critical High: {expected[sensor]['crit_high_th']}}}, "
                            f"Found: {{High: {actual['high_th']}, Critical High: {actual['crit_high_th']}}}")
                status = False

            #     logger.info(f"Matched in {sensor}: expected {expected[sensor]}, got {actual}")
            # else:
            #     status = False

        # Final result
        if status:

        else:
